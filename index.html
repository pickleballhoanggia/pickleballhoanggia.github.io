function addNewPairFromSelection() { const p1 = document.getElementById('doublesNewPlayer1').value.trim(); const p2 = document.getElementById('doublesNewPlayer2').value.trim(); const pointsInput = document.getElementById('doublesAddPoints'); const points = parseInt(pointsInput.value) || 1200; if (!p1 || !p2) { alert("Vui lòng chọn đầy đủ VĐV A và VĐV B!"); return; } if (p1 === p2) { alert("VĐV A và VĐV B không được trùng nhau trong cùng một cặp!"); return; } const pairName1 = `${p1} & ${p2}`; if (!db.doubles) db.doubles = []; // Kiểm tra chặt chẽ: Dùng Set để so sánh tập hợp 2 thành viên bất chấp thứ tự & chữ hoa/thường const newPlayersSet = new Set([p1.toLowerCase(), p2.toLowerCase()]); const exists = db.doubles.some(item => { const parts = item.name.split('&').map(s => s.trim().toLowerCase()); if (parts.length === 2) { const existingSet = new Set(parts); return existingSet.size === newPlayersSet.size && [...newPlayersSet].every(val => existingSet.has(val)); } return item.name.toLowerCase() === pairName1.toLowerCase(); }); if (exists) { alert("Cặp đôi này (hoặc ngược lại vị trí) đã tồn tại trong danh sách hệ thống!"); return; } db.doubles.push({ name: pairName1, points: points, win: 0, loss: 0 }); saveData(); alert(`Đã thêm cặp đôi "${pairName1}" thành công!`); }
